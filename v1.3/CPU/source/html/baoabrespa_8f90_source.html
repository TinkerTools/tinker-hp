121 !
122 ! -> real space

124 !
125 ! communicate positions
126 !
127  call commposrespa(stepfast.ne.nalt)
128  !call commposrespa(.TRUE.)
129 !
130  if(allocated(derivs)) deallocate(derivs)
131  allocate(derivs(3,nbloc))
132  derivs(:,:)=0.d0
133 !

146 !

154  if (use(iglob)) then

159  end do

161 !

163 !

167  if (deb_force) call info_forces(cbond)
168  if (deb_atom) call info_minmax_pva
169 !
170 ! increment average virial from fast-evolving potential terms
171  viralt(:,:)=viralt(:,:)+vir(:,:)/dshort
172 
173  if(ir .and. stepfast<nalt) then
174  !call rotpole
175  call compute_dipole(dip(:,stepfast)&
176  &,dipind(:,stepfast),.false.)
177  endif
178  end do
179 
180  if(use_piston) call apply_a_piston(dt_2,-1,.false.)
181 !
182 ! Reassign the particules that have changed of domain
183 !
184 ! -> real space
185 !
186  time0 = mpi_wtime()
187  call reassignrespa(nalt,nalt)
188 !
189 ! -> reciprocal space

191 !
192 ! communicate positions
193 !
194  call commposrespa(.false.)
195  call commposrec
196  time1 = mpi_wtime()
197  timecommpos = timecommpos + time1 - time0
198 !
199 !
200  time0 = mpi_wtime()
201  call reinitnl(istep)
202  time1 = mpi_wtime()
203  timeinte = timeinte + time1-time0
204  time0 = mpi_wtime()
205 !
206  call mechanic_up_para_respa(istep,.false.)
207 
208  call allocsteprespa(.false.)
209  time1 = mpi_wtime()
210  timeinte = timeinte + time1 - time0
211 !
212 ! rebuild the neighbor lists
213 !
214  time0 = mpi_wtime()
215  if (use_list) call nblist(istep)
216  time1 = mpi_wtime()
217  timenl = timenl + time1 - time0
218 !
219  time0 = mpi_wtime()
220  if(allocated(derivs)) deallocate(derivs)
221  allocate(derivs(3,nbloc))
222  derivs = 0d0
223  time1 = mpi_wtime()

225 !
226 ! get the slow-evolving potential energy and atomic forces
227 !
228  time0 = mpi_wtime()
229  call gradslow(epot,derivs)
230  time1 = mpi_wtime()
231  timegrad = timegrad + time1 - time0
232 !
233 ! communicate forces
234 !
235  time0 = mpi_wtime()
236  call commforcesrespa(derivs,.false.)
237  time1 = mpi_wtime()
238  timecommforces = timecommforces + time1-time0
239 !
240 ! MPI : get total energy
241 !
242  time0 = mpi_wtime()
243  call reduceen(epot)
244  time1 = mpi_wtime()
245  timered = timered + time1 - time0
246 !
247 ! use Newton's second law to get the next accelerations;
248 ! find the full-step velocities using the BAOAB recursion
249 !
250  time0 = mpi_wtime()
251  do i = 1, nloc
252  iglob = glob(i)
253  if (use(iglob)) then
254  a(:,iglob) = -convert * derivs(:,i)/mass(iglob)
255  v(:,iglob) = v(:,iglob) + dt_2*a(:,iglob)
256  end if
257  end do
258 !
259 ! find the constraint-corrected full-step velocities
260 !
261  if (use_rattle) call rattle2(dt)
262 !
263 ! Debug print information
264 !
265  if (deb_energy) call info_energy(rank)
266  if (deb_force) call info_forces(cnbond)
267  if (deb_atom) call info_minmax_pva
268 !
269 ! total potential and virial from sum of fast and slow parts
270 !
271  epot = epot + ealt
272  vir = vir + viralt
273 
274  time1 = mpi_wtime()
275  timeinte = timeinte + time1 - time0
276 !
277  time0 = mpi_wtime()
278  call temper(dt,eksum,ekin,temp)
279  call pressure(dt,ekin,pres,stress,istep)
280  call pressure2(epot,temp)
281  if(ir) then
282  call compute_dipole(dip(:,nalt)&
283  &,dipind(:,nalt),full_dipole)
284  call save_dipole_respa(dip,dipind)
285  endif
286  if(use_piston) then
287  call apply_b_piston(dt,pres,stress)
288  endif
289  time1 = mpi_wtime()
290  timetp = timetp + time1-time0
291 !
292 ! total energy is sum of kinetic and potential energies
293 !
294  time0 = mpi_wtime()
295  etot = eksum + epot
296 !
297 ! compute statistics and save trajectory for this step
298 !
299  call mdsave(istep,dt,epot,derivs)
300  call mdrest(istep)
301  call mdstat(istep,dt,etot,epot,eksum,temp,pres)
302  time1 = mpi_wtime()
303  timeinte = timeinte + time1-time0
304  return
305 end
