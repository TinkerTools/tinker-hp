18 def get_deps_and_mods(filename):
19  deps = []
20  mods = []
21  f = open(filename)
22  if not f:
23  print ("ERROR: unable to open %s%s" % filename)
24  sys.exit(1)
25 #
26 # Match USE statement in fortran sources (case insensitive)
27 #
28  use_line_re = re.compile("(?i)^\s*use\s+(\S.+?)\s*(?:,\s*only\s*:.*)?$")
29 #
30 # Match other modules names in the same USE statement on a continuation line
31 # (i.e. use modulea,
32 # & moduleb
33 #
34  cont_line_re = re.compile("^(.*)&\s*$")
35 #
36 # Match module statement in fortran sources (case insensitive)
37 #
38  mod_line_re = re.compile("(?i)^\s*module\s+(\S+)\s*$")
39 #
40 # Split USE or MODULE statement lines in pieces
41 #
42  split_re = re.compile("\s*,\s*")
43 #
44 # Match any MODULE or USE name
45 # (except MPI and iso_c_binding, which are not in a source file, but come with the compiler)
46 # (except decomp_2d which is external)
47 #
48  dep_re = re.compile("(?i)(^(?!.*mpi|.*iso_c_binding|.*decomp_2d).*$)")
49  mod_re = re.compile("(?i)(^(?!.*mpi|.*iso_c_binding|.*decomp_2d).*$)")
50  within_use_statement = False
51  for line in f:
52  match = use_line_re.search(line)
53  if match:
54  within_use_statement = True
55  rest_line = match.group(1)
56  else:
57  rest_line = line
58  if within_use_statement:
59 #
60 # We're in USE statement
61 #
62  match = cont_line_re.search(rest_line)
63  if match:
64  rest_line = match.group(1)
65  else:
66  within_use_statement = False
67  line_items = split_re.split(rest_line.strip())
68 #
69 # Loop over all modules found in USE statements
70 #
71  for item in line_items:
72  if item:
73  match = dep_re.match(item)
74  if match:
75 #
76 # Add the MOD_file.o corresponding to the modules found. Just one time
77 #
78  dep = match.expand("MOD_\\1.o")
79  if dep not in deps:
80  deps.append(dep)
81  else:
82 #
83 # We're in a MODULE statement
84 #
85  match = mod_line_re.search(line)
86  if match:
87  mod_name = match.group(1)
88  match = mod_re.match(mod_name)
89  if match:
90  mod = match.expand("\\1.mod")
91  if mod not in mods:
92  mods.append(mod)
93  f.close()
94 #
95 # Returns the list of dependencies for each file
96 #
97  return (deps, mods)
