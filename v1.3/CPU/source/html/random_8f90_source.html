247 !> @brief

250 !> @param no params

252  implicit none

254  real*8 random

256 !
257 !

293 !
294 ! literature reference:
295 !
296 ! E. B. Saff and A. B. J. Kuijlaars, "Distributing Many
297 ! Points on a Sphere", The Mathematical Intelligencer,
298 ! 19, 5-11 (1997)
299 !
300 !
301 subroutine sphere (ndot,dot)
302  use math
303  implicit none
304  integer i,ndot
305  real*8 theta,phi
306  real*8 h,phiold
307  real*8 tot,tot1
308  real*8 dot(3,*)
309 !
310 !
311 ! find spherical coordinates then convert to Cartesian
312 !
313  tot = dble(ndot)
314  tot1 = dble(ndot-1)
315  do i = 1, ndot
316  h = -1.0d0 + 2.0d0*dble(i-1)/tot1
317  theta = acos(h)
318  if (i.eq.1 .or. i.eq.ndot) then
319  phi = 0.0d0
320  else
321  phi = mod(phiold+3.6d0/sqrt(tot*(1.0d0-h*h)),2.0d0*pi)
322  end if
323  dot(1,i) = sin(theta) * cos(phi)
324  dot(2,i) = sin(theta) * sin(phi)
325  dot(3,i) = cos(theta)
326  phiold = phi
327  end do
328  return
329 end
