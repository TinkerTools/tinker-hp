85 !

87 !

91 ! initialization to get to 2nd order recursion
92 !
93  temp(2,2) = w

96 ! perform one pass to get to 3rd order recursion
97 !
98  temp(3,3) = 0.5d0 * w * temp(2,2)
99  temp(3,2) = 0.5d0 * ((1.0d0+w)*temp(2,1)+(2.0d0-w)*temp(2,2))
100  temp(3,1) = 0.5d0 * (1.0d0-w) * temp(2,1)
101 !
102 ! compute standard B-spline recursion to desired order
103 !
104  do i = 4, bsorder
105  k = i - 1
106  denom = 1.0d0 / dble(k)
107  temp(i,i) = denom * w * temp(k,k)
108  do j = 1, i-2
109  temp(i,i-j) = denom * ((w+dble(j))*temp(k,i-j-1)&
110  &+(dble(i-j)-w)*temp(k,i-j))
111  end do
112  temp(i,1) = denom * (1.0d0-w) * temp(k,1)
113  end do
114 !
115 ! get coefficients for the B-spline first derivative
116 !
117  k = bsorder - 1
118  temp(k,bsorder) = temp(k,bsorder-1)
119  do i = bsorder-1, 2, -1
120  temp(k,i) = temp(k,i-1) - temp(k,i)
121  end do
122  temp(k,1) = -temp(k,1)
123 !
124 ! get coefficients for the B-spline second derivative
125 !
126  if (level .eq. 4) then
127  k = bsorder - 2
128  temp(k,bsorder-1) = temp(k,bsorder-2)
129  do i = bsorder-2, 2, -1
130  temp(k,i) = temp(k,i-1) - temp(k,i)
131  end do
132  temp(k,1) = -temp(k,1)
133  temp(k,bsorder) = temp(k,bsorder-1)
134  do i = bsorder-1, 2, -1
135  temp(k,i) = temp(k,i-1) - temp(k,i)
136  end do
137  temp(k,1) = -temp(k,1)
138 !
139 ! get coefficients for the B-spline third derivative
140 !
141  k = bsorder - 3
142  temp(k,bsorder-2) = temp(k,bsorder-3)
143  do i = bsorder-3, 2, -1
144  temp(k,i) = temp(k,i-1) - temp(k,i)
145  end do
146  temp(k,1) = -temp(k,1)
147  temp(k,bsorder-1) = temp(k,bsorder-2)
148  do i = bsorder-2, 2, -1
149  temp(k,i) = temp(k,i-1) - temp(k,i)
150  end do
151  temp(k,1) = -temp(k,1)
152  temp(k,bsorder) = temp(k,bsorder-1)
153  do i = bsorder-1, 2, -1
154  temp(k,i) = temp(k,i-1) - temp(k,i)
155  end do
156  temp(k,1) = -temp(k,1)
157  end if
158 !
159 ! copy coefficients from temporary to permanent storage
160 !
161  do i = 1, bsorder
162  do j = 1, level
163  thetai(j,i) = temp(bsorder-j+1,i)
164  end do
165  end do
166  return
167 end
