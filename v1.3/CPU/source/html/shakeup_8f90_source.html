470 !
471 ! "chkangle" tests angles to be constrained for their presence
472 ! in small rings and removes constraints that are redundant
473 !
474 ! note this version correctly handles isolated small rings,
475 ! but should remove one additional redundant constraint for
476 ! each ring fusion
477 !
478 !
479 !> @brief
480 !> tests angles to be constrained for their presence
481 !> in small rings and removes constraints that are redundant
482 !> @param[in] ia: index of the first atom
483 !> @param[in] ib: index of the second atom
484 !> @param[in] ic: index of the third atom
485 subroutine chkangle (ia,ib,ic)
486  use couple
487  use freeze
488  use ring
489  implicit none
490  integer i,j,k
491  integer ia,ib,ic
492  integer id,ie,imin
493  logical remove
494 !
495 !
496 ! all internal angles of 3-membered rings are redundant
497 !
498  remove = .false.
499  if (nring3 .ne. 0) then
500  do i = 1, n12(ia)
501  j = i12(i,ia)
502  if (j .eq. ic) remove = .true.
503  end do
504  end if
505 !
506 ! for 4-membered rings, two internal angles are redundant
507 !
508  if (nring4 .ne. 0) then
509  do i = 1, n12(ia)
510  id = i12(i,ia)
511  if (id .ne. ib) then
512  do j = 1, n12(id)
513  k = i12(j,id)
514  if (k .eq. ic) then
515  imin = min(ia,ib,ic,id)
516  if (ib .eq. imin) remove = .true.
517  if (id .eq. imin) remove = .true.
518  end if
519  end do
520  end if
521  end do
522  end if
523 !
524 ! for 5-membered rings, one internal angle is redundant
525 !
526  if (nring5 .ne. 0) then
527  do i = 1, n12(ia)
528  id = i12(i,ia)
529  if (id.ne.ib .and. id.ne.ic) then
530  do j = 1, n12(ic)
531  ie = i12(j,ic)
532  if (ie.ne.ib .and. ie.ne.ia) then
533  do k = 1, n12(id)
534  if (i12(k,id) .eq. ie) then
535  imin = min(ia,ib,ic,id,ie)
536  if (ib .eq. imin) remove = .true.
537  end if
538  end do
539  end if
540  end do
541  end if
542  end do
543  end if
544 !
545 ! remove the constraint from the list if it is redundant
546 !
547  if (remove) nrat = nrat - 1
548  return
549 end
