14 ! trigonal centers via a Wilson-Decius-Cross or Allinger angle;
15 ! also partitions the energy among the atoms
16 !
17 !
18 !> @brief
19 !> computes the out-of-plane bend potential energy at
20 !> trigonal centers via a Wilson-Decius-Cross or Allinger angle;
21 !> also partitions the energy among the atoms
22 !> @param no params
23 subroutine eopbend3

42  integer ia,ib,ic,id,ibloc
43  real*8 e,angle1,force
44  real*8 cosine
45  real*8 dt,dt2,dt3,dt4
46  real*8 xia,yia,zia
47  real*8 xib,yib,zib
48  real*8 xic,yic,zic
49  real*8 xid,yid,zid
50  real*8 xab,yab,zab
51  real*8 xcb,ycb,zcb
52  real*8 xdb,ydb,zdb
53  real*8 xad,yad,zad
54  real*8 xcd,ycd,zcd

61 !

64 !
65 !
66 ! zero out the out-of-plane bend energy and partitioning
67 !
68  neopb = 0
69  eopb = 0.0d0
70  aeopb = 0.0d0
71  header = .true.
72 !
73 ! calculate the out-of-plane bending energy term
74 !
75  do iopbendloc = 1, nopbendloc
76  iopbend = opbendglob(iopbendloc)
77  i = iopb(iopbend)
78  ia = iang(1,i)
79  ib = iang(2,i)
80  ibloc = loc(ib)
81  ic = iang(3,i)
82  id = iang(4,i)
83  force = opbk(iopbend)
84 !
85 ! decide whether to compute the current interaction
86 !
87  if (use_group) call groups(fgrp,ia,ib,ic,id,0,0)
88  proceed = (use(ia) .or. use(ib) .or.&
89  &use(ic) .or. use(id))
90 !
91 ! get the coordinates of the atoms at trigonal center
92 !
93  if (proceed) then
94  xia = x(ia)
95  yia = y(ia)
96  zia = z(ia)
97  xib = x(ib)
98  yib = y(ib)
99  zib = z(ib)
100  xic = x(ic)
101  yic = y(ic)
102  zic = z(ic)
103  xid = x(id)
104  yid = y(id)
105  zid = z(id)
106 !
107 ! compute the out-of-plane bending angle
108 !
109  xab = xia - xib
110  yab = yia - yib
111  zab = zia - zib
112  xcb = xic - xib
113  ycb = yic - yib
114  zcb = zic - zib
115  xdb = xid - xib
116  ydb = yid - yib
117  zdb = zid - zib
118  xad = xia - xid
119  yad = yia - yid
120  zad = zia - zid
121  xcd = xic - xid
122  ycd = yic - yid
123  zcd = zic - zid
124  if (use_polymer) then
125  call image(xab,yab,zab)
126  call image(xcb,ycb,zcb)
127  call image(xdb,ydb,zdb)
128  call image(xad,yad,zad)
129  call image(xcd,ycd,zcd)
130  end if
131 !
132 ! W-D-C angle between A-B-C plane and B-D vector for D-B<AC
133 !
134  if (opbtyp .eq. 'W-D-C') then
135  rab2 = xab*xab + yab*yab + zab*zab
136  rcb2 = xcb*xcb + ycb*ycb + zcb*zcb
137  cc = rab2*rcb2 - (xab*xcb+yab*ycb+zab*zcb)**2
138 !
139 ! Allinger angle between A-C-D plane and D-B vector for D-B<AC
140 !
141  else if (opbtyp .eq. 'ALLINGER') then
142  rad2 = xad*xad + yad*yad + zad*zad
143  rcd2 = xcd*xcd + ycd*ycd + zcd*zcd
144  cc = rad2*rcd2 - (xad*xcd+yad*ycd+zad*zcd)**2
145  end if
146 !
147 ! find the out-of-plane angle bending energy
148 !
149  ee = xdb*(yab*zcb-zab*ycb) + ydb*(zab*xcb-xab*zcb)&
150  &+ zdb*(xab*ycb-yab*xcb)
151  rdb2 = xdb*xdb + ydb*ydb + zdb*zdb
152  if (rdb2.ne.0.0d0 .and. cc.ne.0.0d0) then
153  bkk2 = rdb2 - ee*ee/cc
154  cosine = sqrt(bkk2/rdb2)
155  cosine = min(1.0d0,max(-1.0d0,cosine))
156  angle1 = radian * acos(cosine)
157  dt = angle1
158  dt2 = dt * dt
159  dt3 = dt2 * dt
160  dt4 = dt2 * dt2
161  e = opbunit * force * dt2&
162  &* (1.0d0+copb*dt+qopb*dt2+popb*dt3+sopb*dt4)
163 !
164 ! scale the interaction based on its group membership
165 !
166  if (use_group) e = e * fgrp
167 !
168 ! increment the total out-of-plane bending energy
169 !
170  neopb = neopb + 1
171  eopb = eopb + e
172  aeopb(ibloc) = aeopb(ibloc) + e
173 !
174 ! print a message if the energy of this interaction is large
175 !
176  huge = (e .gt. 2.0d0)
177  if (debug .or. (verbose.and.huge)) then
178  if (header) then
179  header = .false.
180  write (iout,10)
181 10 format (/,' Individual Out-of-Plane Bend',&
182  &' Interactions :',&
183  &//,' Type',25x,'Atom Names',21x,'Angle',&
184  &6x,'Energy',/)
185  end if
186  write (iout,20) id,name(id),ib,name(ib),ia,&
187  &name(ia),ic,name(ic),angle1,e
188 20 format (' O-P-Bend',2x,4(i7,'-',a3),f11.4,f12.4)
189  end if
190  end if
191  end if
192  end do
193  return
194 end
