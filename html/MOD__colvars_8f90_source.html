89 !

91 !

489 !> @brief
490 !> Prepare all the information necessary to compute and monitor CV-related quantities
491 subroutine prepare_colvars
492  use atmtyp
493  use atoms
494  use boxes
495  use colvars
496  use deriv
497  use domdec
498  use inform
499  use iounit
500  use mdstuf
501  use moldyn
502  use mpi
503  use potent
504  use units
505  implicit none
506  integer i,iglob,iloc,ilocrec,ierr
507  real*8 temp
508 !
509  if (deb_path) write(iout,*), 'prepare_colvars '
510 !
511 !
512  decv =0d0
513  decv_tot = 0d0
514  cv_pos = 0d0
515 
516  do i = 1, ncvatoms
517  iglob = cvatoms_ids(i)
518  iloc = loc(iglob)
519  if (repart(iglob).eq.rank) then
520  cv_pos(1,i) = x(iglob)
521  cv_pos(2,i) = y(iglob)
522  cv_pos(3,i) = z(iglob)
523  end if
524  if ((iloc.gt.0).and.(iloc.le.nbloc)) then
525  decv_tot(1,i) = desum(1,iloc)
526  decv_tot(2,i) = desum(2,iloc)
527  decv_tot(3,i) = desum(3,iloc)
528 !
529 ! for RESPA(1) integrators, also add fast/intermediate forces (through aalt array)
530 !
531  if (mts .and. repart(iglob).eq.rank) then
532  temp = -mass(iglob)/convert
533  decv_tot(1,i) = decv_tot(1,i) +&
534  &temp*(aalt(1,iglob)+aalt2(1,iglob))
535  decv_tot(2,i) = decv_tot(2,i) +&
536  &temp*(aalt(2,iglob)+aalt2(2,iglob))
537  decv_tot(3,i) = decv_tot(3,i) +&
538  &temp*(aalt(3,iglob)+aalt2(3,iglob))
539  end if
540  end if
541 !
542 ! also add reciprocal part of the forces (in a separate array for now)
543 !
544  ilocrec = locrec(iglob)
545  if ((ilocrec.gt.0).and.(ilocrec.le.nlocrec2)) then
546  decv_tot(1,i) = decv_tot(1,i) + decrec(1,ilocrec) +&
547  &demrec(1,ilocrec) + deprec(1,ilocrec) + dedsprec(1,ilocrec)
548  decv_tot(2,i) = decv_tot(2,i) + decrec(2,ilocrec) +&
549  &demrec(2,ilocrec) + deprec(2,ilocrec) + dedsprec(2,ilocrec)
550  decv_tot(3,i) = decv_tot(3,i) + decrec(3,ilocrec) +&
551  &demrec(3,ilocrec) + deprec(3,ilocrec) + dedsprec(3,ilocrec)
552  end if
553  end do
554 !
555  if (rank.eq.0) then
556  call mpi_reduce(mpi_in_place,cv_pos,3*ncvatoms,mpi_real8,&
557  &mpi_sum,0,comm_tinker,ierr)
558  call mpi_reduce(mpi_in_place,decv_tot,3*ncvatoms,mpi_real8,&
559  &mpi_sum,0,comm_tinker,ierr)
560  else
561  call mpi_reduce(cv_pos,cv_pos,3*ncvatoms,mpi_real8,&
562  &mpi_sum,0,comm_tinker,ierr)
563  call mpi_reduce(decv_tot,decv_tot,3*ncvatoms,mpi_real8,&
564  &mpi_sum,0,comm_tinker,ierr)
565  end if
566  if (use_lambdadyn) then
567  delambdae = delambdae + delambdaesave
568  delambdaesave = 0d0
569  delambdav = delambdav + delambdavsave
570  delambdavsave = 0d0
571  if (rank.eq.0) then
572  call mpi_reduce(mpi_in_place,delambdav,1,mpi_real8,mpi_sum,&
573  &0,comm_tinker,ierr)
574  call mpi_reduce(mpi_in_place,delambdae,1,mpi_real8,mpi_sum,&
575  &0,comm_tinker,ierr)
576  else
577  call mpi_reduce(delambdav,delambdav,1,mpi_real8,mpi_sum,&
578  &0,comm_tinker,ierr)
579  call mpi_reduce(delambdae,delambdae,1,mpi_real8,mpi_sum,&
580  &0,comm_tinker,ierr)
581  end if
582  end if
583 end subroutine

585 !> @brief
586 !> Retrieve all the CV-related information (forces due to biases) coming from Colvars
587 subroutine distrib_colvars(derivs)
588  use atoms
589  use colvars
590  use deriv
591  use domdec
592  use inform
593  use iounit
594  use mpi
595  use mutant
596  use potent
597  use virial
598  implicit none
599  real*8 derivs(3,*)
600  real*8 vxx,vxy,vxz,vyy,vyz,vzz
601  integer i,iglob,iloc,ierr
602 !
603  if (deb_path) write(iout,*), 'distrib_colvars '
604 !
605 !
606  call mpi_bcast(decv,3*ncvatoms,mpi_real8,0,comm_tinker,ierr)
607 
608  if (use_lambdadyn) then
609  call mpi_bcast(lambda,1,mpi_real8,0,comm_tinker,ierr)
610  call def_lambdadyn(rank)
611  if (use_osrw) then
612  call mpi_bcast(flambdabias,1,mpi_real8,0,comm_tinker,ierr)
613  do i = 1, nloc
614  dxdelambda(:,i) = dxdelambdae(:,i)*dlambdaelambda +&
615  &dxdelambdav(:,i)*dlambdavlambda
616  derivs(:,i) = derivs(:,i) + flambdabias*dxdelambda(:,i)
617  end do
618  end if
619  end if
620 !
621  do i = 1, ncvatoms
622  iglob = cvatoms_ids(i)
