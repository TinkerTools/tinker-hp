185  a(j,iglob) = -convert * derivs(j,i)/mass(iglob)
186  v(j,iglob) = v(j,iglob) +&
187  &(part2*a(j,iglob)+aalt(j,iglob))*dt_x
188  end do

195  time1 = mpi_wtime()

205 !

207 !

216  time1 = mpi_wtime()
217  timeinte = timeinte + time1-time0
218 !
219 ! perform deallocation of some local arrays
220 !
221  deallocate (derivs)
222  return
223 end
