210  function piqtb_kernel_brieuc(omega,ibead)
211  use bath
212  use units
213  use beads
214  use domdec
215  implicit none
216  real*8, intent(in) :: omega
217  integer, intent(in) :: ibead
218  real*8 :: piqtb_kernel_brieuc
219  real*8 :: u
220  integer :: jbead,iom,iter,jbead_start
221  logical, save :: first_in=.true.
222  integer, save :: nom
223  real*8, allocatable, save :: f(:),us(:)
224  real*8, allocatable :: targ(:),vn(:,:),fp(:)
225  real*8, allocatable :: uns(:,:)
226  real*8 :: umax,un,tmax,nu,b1,err,un1,u12,un2,u2
227  real*8 :: tmp,u1
228  logical :: converged
229 
230  if (piqtb_classical_centroid .and. ibead==1) then
231  piqtb_kernel_brieuc=1.0d0
232  return
233  endif
234 
235  nu=real(nbeads,8)
236 
237  if (first_in) then
238  first_in=.false.
239  if (ranktot==0)&
240  &write(*,*) "--- initializing Brieuc-type PIQTB kernel ---"
241 
242  nom=(3*nseg)/2
243  do while (.true.)
244  umax=(nom-1)*u_fact*domega
245  tmax=nu*tanh(umax/nu)/tanh(umax)
246  do jbead=2,nbeads
247  un=sqrt(umax**2 + (omkpi(jbead)*u_fact)**2)
248  tmax = tmax - (umax*tanh(umax/nu))/(un*tanh(un/nu))
249  enddo
250  if(abs(tmax-1.d0)<1.d-2) exit
251  nom=nom*2
252  enddo
253  if(ranktot==0)&
254  &write(*,*) " PIQTB kernel: nom=",nom," (nseg=",nseg,")"
255 
256  b1=1./nu
257  allocate(f(nom),targ(nom),vn(nom,nbeads),us(nom))
258  allocate(uns(nom,nbeads),fp(nom))
259  f(:)=1.d0
260  vn(1,:)=0.d0
261  us(1)=0.d0
262  if (piqtb_classical_centroid) then
263  u12=(omkpi(2)*u_fact)**2
264  targ(1)=u12/(3*(1-1.d0/nu))
265  do iom=2,nom
266  u=real(iom-1,8)*u_fact*domega
267  us(iom)=u
268  u2=u*u
269  un2=u2 + u12
270  tmp=un2*nu/(u/tanh(u/nu)-1.d0)
271  targ(iom)=tmp*(u/tanh(u)-1.d0)/u2
272  do jbead=3,nbeads
273  un2=u2 + (omkpi(jbead)*u_fact)**2
274  un=sqrt(un2)
275  un1=sqrt(un2-u12)
276  uns(iom,jbead)=un1
277  vn(iom,jbead)=tmp*(un1/tanh(un1/nu)-1.d0)/(nu*un2)
278  enddo
279  enddo
280  vn(1,:)=vn(2,:)
281  targ(:)=targ(:)*(nu-1.d0)/nu
282  else
283  targ(1)=1.d0
284  do iom=2,nom
285  u=real(iom-1,8)*u_fact*domega
286  us(iom)=u
287  targ(iom)=nu*tanh(u/nu)/tanh(u)
288  do jbead=2,nbeads
289  un=sqrt(u**2 + (omkpi(jbead)*u_fact)**2)
290  uns(iom,jbead)=un
291  vn(iom,jbead)=(u*tanh(u/nu))/(un*tanh(un/nu))
292  enddo
293  enddo
294  endif
295  converged=.false.
296  jbead_start=2
297  if (piqtb_classical_centroid) jbead_start=3
298  do iter=1,200
299  do iom=1,nom
300  fp(iom)=0.d0
301  do jbead=jbead_start,nbeads
302  fp(iom)=fp(iom)+vn(iom,jbead)&
303  &*linear_interpolation(uns(iom,jbead),us,f,f(1),f(nom))
304  enddo
305  enddo
306  err = maxval(abs(f+fp-targ)/targ)
307  if (err<1.d-3) then
308  if (ranktot==0) then
309  write(*,*) " PIQTB kernel: converged in "&
310  &,iter," iterations"
311  endif
312  converged=.true.
313  exit
314  endif
315  f = b1*(targ-fp) + (1.d0-b1)*f
316  enddo
317  if (.not.converged .and. ranktot==0) then
318  write(*,*) " PIQTB kernel: not converged !"
319  call fatal
320  endif
321 
322  deallocate(targ,vn,uns,fp)
323  endif
324 
325  if(omega*u_fact<us(2)) then
326  piqtb_kernel_brieuc=1.0d0
327  return
328  endif
329 
330  u = omega*u_fact
331  if(piqtb_classical_centroid) then
332  u12=u*u-(omkpi(2)*u_fact)**2
333  if(u12<=0.d0) then
334  piqtb_kernel_brieuc=1.0d0
335  return
336  endif
337  u1=sqrt(u12)
338  piqtb_kernel_brieuc = (u1/tanh(u1/nu)-1.d0)/(nu-1.d0)&
339  &*linear_interpolation(u1,us,f,1.d0,f(nom))
340  else
341  piqtb_kernel_brieuc = u/(nu*tanh(u/nu))&
342  &*linear_interpolation(u,us,f,1.d0,f(nom))
343  endif
344 
345  end function piqtb_kernel_brieuc

596 ! !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
597 ! Following is a FFT implementation based on algorithm proposed by
598 ! Glassman, a general FFT algorithm supporting arbitrary input length.
599 !
600 ! W. E. Ferguson, Jr., "A simple derivation of Glassman general-n fast
601 ! Fourier transform," Comput. and Math. with Appls., vol. 8, no. 6, pp.
602 ! 401-411, 1982.
603 !
604 ! Original implemtation online at http://www.jjj.de/fft/fftpage.html
605 !
606 ! Updated
607 ! - to handle double-precision as well
608 ! - unnecessary scaling code removed
609 ! !!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
610 
611  SUBROUTINE spcfft(U,N,ISIGN,WORK)
612 
613  IMPLICIT NONE
614 
615  LOGICAL :: inu
616  INTEGER :: a,b,c,n,i,isign
617  COMPLEX(FFT1D_s) :: u(*),work(*)
618 
619  a = 1
620  b = n
621  c = 1
622  inu = .true.
623 
624  DO WHILE ( b .GT. 1 )
625  a = c * a
626  c = 2
627  DO WHILE ( mod(b,c) .NE. 0 )
628  c = c + 1
629  END DO
630  b = b / c
631  IF ( inu ) THEN
632  CALL spcpft(a,b,c,u,work,isign)
633  ELSE
634  CALL spcpft(a,b,c,work,u,isign)
635  END IF
636  inu = ( .NOT. inu )
637  END DO
638 
639  IF ( .NOT. inu ) THEN
640  DO i = 1, n
641  u(i) = work(i)
642  END DO
643  END IF
644 
645  RETURN
646  END SUBROUTINE spcfft

649  SUBROUTINE spcpft( A, B, C, UIN, UOUT, ISIGN )
650 
651  IMPLICIT NONE
652 
653  INTEGER :: isign,a,b,c,ia,ib,ic,jcr,jc
654 
655  DOUBLE PRECISION :: angle
656 
657  COMPLEX(FFT1D_s) :: uin(b,c,a),uout(b,a,c),delta,omega,sum
658 
659  angle = 6.28318530717958_fft1d_s / REAL( a * c, kind=fft1d_s)
660  omega = cmplx( 1.0, 0.0, kind=fft1d_s )
661 
662  IF( isign .EQ. 1 ) THEN
