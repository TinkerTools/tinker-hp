118 !
119 ! given the index ii of a multipole, gives as output the indexes of
120 ! the atoms used to define the local frame and the derivatives of
121 ! the rotation matrix with respect to the position of such atoms.
122 !

124  real*8 r(3,3), dri(3,3,3), driz(3,3,3), drix(3,3,3), driy(3,3,3)
125  logical doder
126  integer j, k, l
127  real*8 xxi, xxi2, eet, eet2, zze, zze2, uu, uu2, uu3,&
128  &vez, eex, eex2, zero, one
129  real*8 xi(3), eta(3), zeta(3), u(3), v(3), ez(3), ex(3), ey(3)
130  real*8 ezu(3,3), exv(3,3), exez(3,3), eyez(3,3), eyex(3,3),&

132  &vriy(3,3), exi(3,3), exiz(3,3), exix(3,3), exiy(3,3), eyi(3,3),&

139 1000 format('polaxe not recognized or not implemented.')
140 !
141 ! Get the reference atoms coordinates.
142 !
143  i = ipole(ii)
144 !
145  if (polaxe(ii).eq.'None') then
146  r = zero
147  dri = zero
148  driz = zero
149  drix = zero
150  driy = zero
151  return
152  end if
153 !
154  iz = zaxis(ii)
155  if (iz.le.0) iz = i
156  ix = xaxis(ii)
157  if (ix.le.0) ix = i
158  iy = yaxis(ii)
159  if (iy.le.0) iy = i
160 
161  if (iz.ne.0) then
162  xi(1) = x(iz) - x(ii)
163  xi(2) = y(iz) - y(ii)
164  xi(3) = z(iz) - z(ii)
165  xxi2 = xi(1)*xi(1) + xi(2)*xi(2) + xi(3)*xi(3)
166  xxi = sqrt(xxi2)
167  end if
168  if (ix.ne.0) then
169  eta(1) = x(ix) - x(ii)
170  eta(2) = y(ix) - y(ii)
171  eta(3) = z(ix) - z(ii)
172  eet2 = eta(1)*eta(1) + eta(2)*eta(2) + eta(3)*eta(3)
173  eet = sqrt(eet2)
174  end if
175  if (iy.ne.0) then
176  zeta(1) = x(iy) - x(ii)
177  zeta(2) = y(iy) - y(ii)
178  zeta(3) = z(iy) - z(ii)
179  zze2 = zeta(1)*zeta(1) + zeta(2)*zeta(2) + zeta(3)*zeta(3)
180  zze = sqrt(zze2)
181  end if
182 !
183 ! We will write ez = u/|u|, ex = (v - (v*ez)ez)/(v - (v*ez)ez) and
184 ! ey = ez x ex for all the possible definitions.
185 ! Here, we define for each method the u and v vectors:
186 !
187  if (polaxe(i).eq.'Z-then-X') then
188  u(1) = xi(1)
189  u(2) = xi(2)
190  u(3) = xi(3)
191  v(1) = eta(1)
192  v(2) = eta(2)
193  v(3) = eta(3)
194  else if (polaxe(i).eq.'Bisector') then
195  u(1) = eet*xi(1) + xxi*eta(1)
196  u(2) = eet*xi(2) + xxi*eta(2)
197  u(3) = eet*xi(3) + xxi*eta(3)
198  v(1) = eta(1)
199  v(2) = eta(2)
200  v(3) = eta(3)
201  else if (polaxe(i).eq.'Z-Bisect') then
202  u(1) = xi(1)
203  u(2) = xi(2)
204  u(3) = xi(3)
205  v(1) = zze*eta(1) + eet*zeta(1)
206  v(2) = zze*eta(2) + eet*zeta(2)
207  v(3) = zze*eta(3) + eet*zeta(3)
208  else if (polaxe(i).eq.'3-Fold') then
209  u(1) = eet*zze*xi(1) + xxi*zze*eta(1) + eet*zze*zeta(1)
210  u(2) = eet*zze*xi(2) + xxi*zze*eta(2) + eet*zze*zeta(2)
211  u(3) = eet*zze*xi(3) + xxi*zze*eta(3) + eet*zze*zeta(3)
212  v(1) = eta(1)
213  v(2) = eta(2)
214  v(3) = eta(3)
215  else
216  write(6,1000)
217  call fatal
218  end if
219  uu2 = u(1)*u(1) + u(2)*u(2) + u(3)*u(3)
220  uu = sqrt(uu2)
221 !
222 ! Assemble the three versors:
223 !
224  ez(1) = u(1)/uu
225  ez(2) = u(2)/uu
226  ez(3) = u(3)/uu
227  vez = ez(1)*v(1) + ez(2)*v(2) + ez(3)*v(3)
228  ex(1) = v(1) - vez*ez(1)
229  ex(2) = v(2) - vez*ez(2)
230  ex(3) = v(3) - vez*ez(3)
231  eex2 = ex(1)*ex(1) + ex(2)*ex(2) + ex(3)*ex(3)
232  eex = sqrt(eex2)
233  ex(1) = ex(1)/eex
234  ex(2) = ex(2)/eex
235  ex(3) = ex(3)/eex
236  ey(1) = ez(2)*ex(3) - ez(3)*ex(2)
237  ey(2) = ez(3)*ex(1) - ez(1)*ex(3)
238  ey(3) = ez(1)*ex(2) - ez(2)*ex(1)
239  do j = 1, 3
240  r(j,1) = ex(j)
241  r(j,2) = ey(j)
242  r(j,3) = ez(j)
243  end do
244  if (.not. doder) return
245 !
246 ! clear everything.
247 !
248  do j = 1, 3
249  do k = 1, 3
250  ezu(j,k) = zero
251  exv(j,k) = zero
252  exez(j,k) = zero
253  eyez(j,k) = zero
254  eyex(j,k) = zero
255  uri(j,k) = zero
256  uriz(j,k) = zero
257  urix(j,k) = zero
258  uriy(j,k) = zero
259  vri(j,k) = zero
260  vrix(j,k) = zero
261  vriy(j,k) = zero
262  exi(j,k) = zero
263  exiz(j,k) = zero
264  exix(j,k) = zero
265  exiy(j,k) = zero
266  eyi(j,k) = zero
267  eyiz(j,k) = zero
268  eyix(j,k) = zero
269  eyiy(j,k) = zero
270  ezi(j,k) = zero
271  eziz(j,k) = zero
272  ezix(j,k) = zero
273  eziy(j,k) = zero
274  do l = 1, 3
275  dri(j,k,l) = zero
276  driz(j,k,l) = zero
277  drix(j,k,l) = zero
278  driy(j,k,l) = zero
279  end do
280  end do
281  end do
282 !
283 ! We will assemble the derivatives of the rotation matrices as
284 ! the product of two contributions, according to the chain rule.
285 ! The first part is the derivative of the versors wrt the u, v
286 ! vectors and is independent of the model; the second part is the
287 ! derivatives of u and v wrt the positions of the involved atoms
288 ! and is specific. We will assemble here the first part. For later
289 ! convenience we will also compute here dex/dez, dey/dez and
290 ! dey/dex.
291 !
292  uu3 = uu*uu2
293  do j = 1, 3
294  ezu(j,j) = one/uu
295  exv(j,j) = one/eex
296  exez(j,j) = -vez/eex
297  do k = 1, 3
298  ezu(j,k) = ezu(j,k) - u(j)*u(k)/uu3
299  exv(j,k) = exv(j,k) - ez(j)*ez(k)/eex - ex(j)*ex(k)/eex
300  exez(j,k) = exez(j,k) + ex(j)*vez*v(k)/eex2 - ez(j)*v(k)/eex
301  end do
302  end do
303  eyez(1,1) = zero
304  eyez(1,2) = ex(3)
305  eyez(1,3) = -ex(2)
306  eyez(2,1) = -ex(3)
307  eyez(2,2) = zero
308  eyez(2,3) = ex(1)
309  eyez(3,1) = ex(2)
310  eyez(3,2) = -ex(1)
311  eyez(3,3) = zero
312  eyex(1,1) = zero
313  eyex(1,2) = -ez(3)
314  eyex(1,3) = ez(2)
315  eyex(2,1) = ez(3)
316  eyex(2,2) = zero
317  eyex(2,3) = -ez(1)
318  eyex(3,1) = -ez(2)
319  eyex(3,2) = ez(1)
320  eyex(3,3) = zero
321 !
322 ! We compute now all the chain rule contributions, in particular:
323 ! du/dr(i), du/dr(iz), du/dr(ix), du/dr(iy)
324 ! dv/dr(i), dv/dr(ix), dv/dr(iy)
325 ! which we will use to assemble the derivatives of the rotation
326 ! matrix.
327 !
328  if (polaxe(ii).eq.'Z-then-X') then
329  do j = 1, 3
330  uri(j,j) = -one
331  uriz(j,j) = one
332  vri(j,j) = -one
333  vrix(j,j) = one
334  end do
335  else if (polaxe(ii).eq.'Bisector') then
336  do j = 1, 3
337  uri(j,j) = -xxi - eet
338  uriz(j,j) = eet
339  urix(j,j) = xxi
340  vri(j,j) = -one
341  vrix(j,j) = one
342  do k = 1, 3
343  uri(j,k) = uri(j,k) - eta(j)*xi(k)/xxi - xi(j)*eta(k)/eet
344  uriz(j,k) = uriz(j,k) + eta(j)*xi(k)/xxi
345  urix(j,k) = urix(j,k) + xi(j)*eta(k)/eet
346  end do
347  end do
348  else if (polaxe(ii).eq.'Z-Bisect') then
349  do j = 1, 3
350  uri(j,j) = -one
351  uriz(j,j) = one
352  vri(j,j) = -eet - zze
353  vrix(j,j) = zze
354  vriy(j,j) = eet
355  do k = 1, 3
356  vri(j,k) = vri(j,k) - eta(j)*zeta(k)/zze&
357  &- zeta(j)*eta(k)/eet
358  vrix(j,k) = vrix(j,k) + zeta(j)*eta(k)/eet
359  vriy(j,k) = vriy(j,k) + eta(j)*zeta(k)/zze
360  end do
361  end do
362  else if (polaxe(ii).eq.'3-Fold') then
363  do j = 1, 3
364  uri(j,j) = -xxi*eet - xxi*zze - eet*zze
365  uriz(j,j) = +eet*zze
366  urix(j,j) = +xxi*zze
367  uriy(j,j) = +xxi*eet
368  vri(j,j) = -one
369  vrix(j,j) = one
370  do k = 1, 3
371  uri(j,k) = uri(j,k) - (zze*eta(j)+eet*zeta(j))*xi(k)/xxi&
372  &- (zze*xi(j)+xxi*zeta(j))*eta(k)/eet&
373  &- (eet*xi(j)+xxi*eta(j))*zeta(k)/zze
374  uriz(j,k) = uriz(j,k) + (zze*eta(j)+eet*zeta(j))*xi(k)/xxi
375  urix(j,k) = urix(j,k) + (zze*xi(j)+xxi*zeta(j))*eta(k)/eet
376  uriy(j,k) = uriy(j,k) + (eet*xi(j)+xxi*eta(j))*zeta(k)/zze
377  end do
378  end do
379  else
380  write(6,1000)
381  call fatal
382  end if
383 !
384 ! We are, finally, ready to assemble the derivatives of each versor.
385 ! The code does not make distinctions anymore between the various
386 ! methods to define the local frame. This is not top efficient, but
387 ! the price to pay for a more efficient code would be a very messy
388 ! one: as the overall cost of the computation is small, order is
389 ! preferred.
390 !
391 ! ez derivatives:
392 !
393  do l = 1, 3
394  do j = 1, 3
395  do k = 1, 3
396  ezi(l,j) = ezi(l,j) + ezu(l,k)*uri(k,j)
397  eziz(l,j) = eziz(l,j) + ezu(l,k)*uriz(k,j)
398  ezix(l,j) = ezix(l,j) + ezu(l,k)*urix(k,j)
399  eziy(l,j) = eziy(l,j) + ezu(l,k)*uriy(k,j)
400  end do
401  end do
402  end do
403 !
404 ! ex derivatives:
405 !
406  do l = 1, 3
407  do j = 1, 3
408  do k = 1, 3
409  exi(l,j) =exi(l,j) +exv(l,k)*vri(k,j) +exez(l,k)*ezi(k,j)
410  exiz(l,j)=exiz(l,j) +exez(l,k)*eziz(k,j)
411  exix(l,j)=exix(l,j)+exv(l,k)*vrix(k,j)+exez(l,k)*ezix(k,j)
412  exiy(l,j)=exiy(l,j)+exv(l,k)*vriy(k,j)+exez(l,k)*eziy(k,j)
413  end do
414  end do
415  end do
416 !
417 ! ey derivatives:
418 !
419  do l = 1, 3
420  do j = 1, 3
421  do k = 1, 3
422  eyi(l,j) =eyi(l,j) +eyex(l,k)*exi(k,j) +eyez(l,k)*ezi(k,j)
423  eyiz(l,j)=eyiz(l,j)+eyex(l,k)*exiz(k,j)+eyez(l,k)*eziz(k,j)
424  eyix(l,j)=eyix(l,j)+eyex(l,k)*exix(k,j)+eyez(l,k)*ezix(k,j)
425  eyiy(l,j)=eyiy(l,j)+eyex(l,k)*exiy(k,j)+eyez(l,k)*eziy(k,j)
426  end do
427  end do
428  end do
429 !
430 ! Finally, assemble the derivatives of the rotation matrix.
431 !
432  do j = 1, 3
433  do k = 1, 3
434  dri(k,j,1) = exi(j,k)
435  dri(k,j,2) = eyi(j,k)
436  dri(k,j,3) = ezi(j,k)
437  driz(k,j,1) = exiz(j,k)
438  driz(k,j,2) = eyiz(j,k)
439  driz(k,j,3) = eziz(j,k)
440  drix(k,j,1) = exix(j,k)
441  drix(k,j,2) = eyix(j,k)
442  drix(k,j,3) = ezix(j,k)
443  driy(k,j,1) = exiy(j,k)
444  driy(k,j,2) = eyiy(j,k)
445  driy(k,j,3) = eziy(j,k)
446  end do
447  end do
448  return
449 end
