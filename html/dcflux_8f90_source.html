4 ! ## subroutine dcflux -- charge flux gradient chain rule ##
5 ! ## ##

8 !
9 ! "dcflux" takes as input the electrostatic potential at each
10 ! atomic site and calculates gradient chain rule terms due to
11 ! charge flux coupling with bond stretching and angle bending
12 !
13 ! literature reference:
14 !
15 ! C. Liu, J.-P. Piquemal and P. Ren, "Implementation of Geometry-
16 ! Dependent Charge Flux into the Polarizable AMOEBA+ Potential",
17 ! Journal of Physical Chemistry Letters, 11, 419-426 (2020)
18 !
19 !
20 !> @brief
21 !> takes as input the electrostatic potential at each
22 !> atomic site and calculates gradient chain rule terms due to
23 !> charge flux coupling with bond stretching and angle bending
24 !> @param[in] pot: electrostatic potential at each site
25 !> @param[in] dcfx: derivative contribution on x coordinate
26 !> @param[in] dcfy: derivative contribution on y coordinate
27 !> @param[in] dcfz: derivative contribution on z coordinate
28 subroutine dcflux (pot,dcfx,dcfy,dcfz)

40  integer i,ia,ib,ic
41  integer ialoc,ibloc,icloc
42  integer iangle,ibond
43  real*8 xa,ya,za
44  real*8 xb,yb,zb
45  real*8 xc,yc,zc
46  real*8 xab,yab,zab
47  real*8 xcb,ycb,zcb
48  real*8 rab,rab2,rab3
49  real*8 rcb,rcb2,rcb3
50  real*8 dpot,dpota,dpotc

52  real*8 fx,fy,fz

55  real*8 fxc1,fyc1,fzc1

57  real*8 fxb2,fyb2,fzb2

68  real*8 dcfy(*)
69  real*8 dcfz(*)
70 !
71 !
72  if (deb_path) write(iout,*), 'dcflux '
73 !
74 !
75 ! zero out the charge flux correction forces
76 !
77  do i = 1, nbloc
78  dcfx(i) = 0.0d0
79  dcfy(i) = 0.0d0
80  dcfz(i) = 0.0d0
81  end do
82 !
83 ! set tolerance for minimum distance and angle values
84 !
85  eps = 0.0001d0
86 !
87 ! calculate the charge flux forces due to bond stretches
88 !
89  do ibond = 1, nbondloc
90  i = bndglob(ibond)
91  ia = ibnd(1,i)
92  ialoc = loc(ia)
93  ib = ibnd(2,i)
94  ibloc = loc(ib)
95  pb = bflx(i)
96  xa = x(ia)
97  ya = y(ia)
98  za = z(ia)
99  xb = x(ib)
100  yb = y(ib)
101  zb = z(ib)
102  xab = xa - xb
103  yab = ya - yb
104  zab = za - zb
105  if (use_polymer) call image(xab,yab,zab)
106  rab2 = max(xab*xab+yab*yab+zab*zab,eps)
107  pb = pb / sqrt(rab2)
108  dpot = pot(ibloc) - pot(ialoc)
109  ddqdx = pb * (xa-xb)
110  ddqdy = pb * (ya-yb)
111  ddqdz = pb * (za-zb)
112  fx = dpot * ddqdx
113  fy = dpot * ddqdy
114  fz = dpot * ddqdz
115  dcfx(ialoc) = dcfx(ialoc) + fx
116  dcfy(ialoc) = dcfy(ialoc) + fy
117  dcfz(ialoc) = dcfz(ialoc) + fz
118  dcfx(ibloc) = dcfx(ibloc) - fx
119  dcfy(ibloc) = dcfy(ibloc) - fy
120  dcfz(ibloc) = dcfz(ibloc) - fz
121  end do
122 !
123 ! calculate the charge flux forces due to angle bends
124 !
125  do iangle = 1, nangleloc
126  i = angleglob(iangle)
127  ia = iang(1,i)
128  ialoc = loc(ia)
129  ib = iang(2,i)
130  ibloc = loc(ib)
131  ic = iang(3,i)
132  icloc = loc(ic)
133  pa1 = aflx(1,i)
134  pa2 = aflx(2,i)
135  pb1 = abflx(1,i)
136  pb2 = abflx(2,i)
137  xa = x(ia)
138  ya = y(ia)
139  za = z(ia)
140  xb = x(ib)
141  yb = y(ib)
142  zb = z(ib)
143  xc = x(ic)
144  yc = y(ic)
145  zc = z(ic)
146  xab = xa - xb
147  yab = ya - yb
148  zab = za - zb
149  xcb = xc - xb
150  ycb = yc - yb
151  zcb = zc - zb
152  if (use_polymer) then
153  call image(xab,yab,zab)
154  call image(xcb,ycb,zcb)
155  end if
156  rab2 = max(xab*xab+yab*yab+zab*zab,eps)
157  rcb2 = max(xcb*xcb+ycb*ycb+zcb*zcb,eps)
158  rab = sqrt(rab2)
159  rab3 = rab2 * rab
160  rcb = sqrt(rcb2)
161  rcb3 = rcb2 * rcb
162 !
163 ! get terms corresponding to asymmetric bond stretches
164 !
165  dpota = pot(ialoc) - pot(ibloc)
166  dpotc = pot(icloc) - pot(ibloc)
167  pb1 = dpota * pb1
168  pb2 = dpotc * pb2
169  fxa1 = pb2 * xab/rab
170  fya1 = pb2 * yab/rab
171  fza1 = pb2 * zab/rab
172  fxc1 = pb1 * xcb/rcb
173  fyc1 = pb1 * ycb/rcb
174  fzc1 = pb1 * zcb/rcb
175  fxb1 = -fxa1 - fxc1
176  fyb1 = -fya1 - fyc1
177  fzb1 = -fza1 - fzc1
178 !
179 ! get terms corresponding to bond angle bending
180 !
181  dot = xab*xcb + yab*ycb + zab*zcb
182  term = -rab*rcb / max(sqrt(rab2*rcb2-dot*dot),eps)
183  fterm = term * (dpota*pa1+dpotc*pa2)
184  termxa = xcb/(rab*rcb) - xab*dot/(rab3*rcb)
185  termya = ycb/(rab*rcb) - yab*dot/(rab3*rcb)
186  termza = zcb/(rab*rcb) - zab*dot/(rab3*rcb)
187  termxc = xab/(rab*rcb) - xcb*dot/(rab*rcb3)
188  termyc = yab/(rab*rcb) - ycb*dot/(rab*rcb3)
189  termzc = zab/(rab*rcb) - zcb*dot/(rab*rcb3)
190  fxa2 = fterm * termxa
191  fya2 = fterm * termya
192  fza2 = fterm * termza
193  fxc2 = fterm * termxc
194  fyc2 = fterm * termyc
195  fzc2 = fterm * termzc
196  fxb2 = -fxa2 - fxc2
197  fyb2 = -fya2 - fyc2
198  fzb2 = -fza2 - fzc2
199  dcfx(ialoc) = dcfx(ialoc) + fxa1 + fxa2
200  dcfy(ialoc) = dcfy(ialoc) + fya1 + fya2
201  dcfz(ialoc) = dcfz(ialoc) + fza1 + fza2
202  dcfx(ibloc) = dcfx(ibloc) + fxb1 + fxb2
203  dcfy(ibloc) = dcfy(ibloc) + fyb1 + fyb2
204  dcfz(ibloc) = dcfz(ibloc) + fzb1 + fzb2
205  dcfx(icloc) = dcfx(icloc) + fxc1 + fxc2
206  dcfy(icloc) = dcfy(icloc) + fyc1 + fyc2
207  dcfz(icloc) = dcfz(icloc) + fzc1 + fzc2
208  end do
209  return
210 end
